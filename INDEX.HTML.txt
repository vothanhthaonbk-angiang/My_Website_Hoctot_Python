    async function initializePython() {
      try {
        statusEl.textContent = "Đang tải Pyodide; lần đầu có thể mất một lúc…";
        pyodide = await loadPyodide();
        statusEl.textContent = "Sẵn sàng! Bạn có thể chạy chương trình Python.";
      } catch (error) {
        statusEl.textContent = "Không tải được Pyodide. Hãy kiểm tra Internet hoặc thử lại.";
        outputEl.textContent = String(error);
      }
    }

    runButton.addEventListener("click", async () => {
      if (running) return;
      if (!pyodide) {
        outputEl.textContent = "Trình chạy chưa sẵn sàng. Hãy chờ tải Pyodide hoàn tất.";
        return;
      }
      running = true;
      runButton.disabled = true;
      outputEl.textContent = "Đang chạy…";
      try {
        // Ví dụ đơn giản: chuyển print() thành vùng ghi kết quả của website.
        // input() dùng hộp thoại prompt của trình duyệt.
        pyodide.setStdout({ batched: (text) => {
          outputEl.textContent = (outputEl.textContent === "Đang chạy…" ? "" : outputEl.textContent) + text + "\\n";
        }});
        pyodide.setStderr({ batched: (text) => {
          outputEl.textContent = (outputEl.textContent === "Đang chạy…" ? "" : outputEl.textContent) + text + "\\n";
        }});
        pyodide.globals.set("browser_input", (promptText) => {
          const answer = window.prompt(String(promptText ?? "Nhập dữ liệu:"));
          if (answer === null) throw new Error("Bạn đã hủy thao tác nhập dữ liệu.");
          return answer;
        });
        const wrappedCode = "input = browser_input\\n" + codeEl.value;
        await pyodide.runPythonAsync(wrappedCode);
        if (!outputEl.textContent.trim()) outputEl.textContent = "Chương trình chạy xong nhưng không có nội dung in ra.";
        statusEl.textContent = "Chạy chương trình hoàn tất.";
      } catch (error) {
        outputEl.textContent += "\\nLỗi: " + String(error);
        statusEl.textContent = "Chương trình gặp lỗi. Hãy đọc thông báo và sửa mã.";
      } finally {
        running = false;
        runButton.disabled = false;
      }
    });

    });

      });
